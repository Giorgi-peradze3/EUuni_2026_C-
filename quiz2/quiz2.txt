class Product {
private:
    string name;
    string category;
    double price;

public:
    
    Product(string name, string category, double price) {
        this->name = name;
        this->category = category;
        this->price = price;
    }

    
    string getName() { 
        return name; 
        
    }
    string getCategory() { 
        return category; 
        
    }
    double getPrice() { 
        return price; 
        
    }
};



class Order {
private:
    
    vector<pair<Product, int>> items;
    
    
    double totalAmount;

   
    void calculateTotalAmount() {
        totalAmount = 0;
        for (const auto& item : items) {
            Product prod = item.first;
            int quantity = item.second;
            totalAmount += prod.getPrice() * quantity;
        }
    }

public:
    
    Order(vector<pair<Product, int>> orderItems) {
        items = orderItems;
        
        calculateTotalAmount();
    }

    
    double getTotalAmount() const {
        return totalAmount;
    }

    
    void printOrder() const {
        cout << "============= ORDER DETAILS =============" << endl;
        cout << "Purchased Products:" << endl;
        
        for (const auto& item : items) {
            Product prod = item.first;
            int quantity = item.second;
            
            cout << " - " << prod.getName() 
                 << " [" << prod.getCategory() << "] | "
                 << "Price: " << prod.getPrice() << " | "
                 << "Quantity: " << quantity << " | "
                 << "Subtotal: " << prod.getPrice() * quantity << endl;
        }
        
        cout << "-----------------------------------------" << endl;
        cout << "Total Amount Due: " << totalAmount << endl;
        cout << "=========================================" << endl;
    }
};

int main() {
    
    Product p1("Laptop", "Electronics", 1500.0);
    Product p2("Mouse", "Electronics", 50.0);
    Product p3("C++ Book", "Education", 35.0);

    
    vector<pair<Product, int>> myCart;
    myCart.push_back({p1, 1}); 
    myCart.push_back({p2, 2}); 
    myCart.push_back({p3, 3}); 

   
    Order myOrder(myCart);

   
    myOrder.printOrder();

    return 0;
}

class bankaccount {
private:
    string accountnumber;
    double balance;
        
public:
    bankaccount(string accnum, double bal) {
        accountnumber = accnum;
        balance = bal;
    }
        
    string getaccountnumber() {
        return accountnumber;
    }
        
    
    double getbalance() {
        return balance;
    }
        
    void deposit(double amount) {
        if (amount > 0) {
            balance += amount;
            cout << "Successfully deposited: " << amount << endl;
        } 
    } 
        
    void withdraw(double amount) {
        if (amount > balance) {
            cout << "Error: Insufficient funds! Current balance is " << balance << endl;
        } else {
            balance -= amount;
            cout << "Successfully withdrew: " << amount << endl;
        }
    } // Added missing closing brace

    void displaysummary() const {
        cout << "---------------------------------" << endl;
        cout << "Account Number: " << accountnumber << endl; 
        cout << "Current Balance: " << balance << endl;
        cout << "---------------------------------" << endl;
    }
}; 

int main() { 
    
    bankaccount myacc("h34r-rty6", 7685);
    
    cout << "Initial Account Status:" << endl;
    myacc.displaysummary();
    
    myacc.deposit(500);
    cout << "Balance after deposit: " << myacc.getbalance() << endl;
    
    myacc.withdraw(200);
    myacc.withdraw(654);
    
    cout << "\nFinal Account Status:" << endl;
    myacc.displaysummary();
    
    return 0;
}

class Shape {
public:
    
    virtual double calculateArea() const = 0;
    virtual double calculatePerimeter() const = 0;

    
    virtual ~Shape() {}
};


class Circle : public Shape {
private:
    double radius;

public:
    Circle(double r) {
        radius = r > 0 ? r : 0;
    }

  
    double calculateArea() const override {
        return M_PI * radius * radius;
    }

    double calculatePerimeter() const override {
        return 2 * M_PI * radius;
    }
};


class Rectangle : public Shape {
private:
    double width;
    double height;

public:
    Rectangle(double w, double h) {
        width = w > 0 ? w : 0;
        height = h > 0 ? h : 0;
    }

    double calculateArea() const override {
        return width * height;
    }

    double calculatePerimeter() const override {
        return 2 * (width + height);
    }
};


class Triangle : public Shape {
private:
    double sideA, sideB, sideC;

public:
    Triangle(double a, double b, double c) {
        
        if (a + b > c && a + c > b && b + c > a) {
            sideA = a;
            sideB = b;
            sideC = c;
        } else {
            cout << "Warning: Invalid triangle sides! Setting sides to 0." << endl;
            sideA = sideB = sideC = 0;
        }
    }

    
    double calculateArea() const override {
        double s = (sideA + sideB + sideC) / 2.0; 
        return sqrt(s * (s - sideA) * (s - sideB) * (s - sideC));
    }

    double calculatePerimeter() const override {
        return sideA + sideB + sideC;
    }
};


int main() {
    
    Shape* circle = new Circle(5.0);          
    Shape* rectangle = new Rectangle(4.0, 6.0); 
    Shape* triangle = new Triangle(3.0, 4.0, 5.0); 

    
    cout << "--- Circle Details ---" << endl;
    cout << "Area: " << circle->calculateArea() << endl;
    cout << "Perimeter: " << circle->calculatePerimeter() << endl << endl;

    
    cout << "--- Rectangle Details ---" << endl;
    cout << "Area: " << rectangle->calculateArea() << endl;
    cout << "Perimeter: " << rectangle->calculatePerimeter() << endl << endl;

    
    cout << "--- Triangle Details ---" << endl;
    cout << "Area: " << triangle->calculateArea() << endl;
    cout << "Perimeter: " << triangle->calculatePerimeter() << endl << endl;


    return 0;
}
